// Neutrální přepínač porovnání; žádné produkční služby ani zápisy.
const scenarios=[['identity','Ikony a lišta','Porovnej ikonu v Docku a malé stavové ikony na světlé i tmavé liště.'],['detail','Detail nahrávky','Poznáš, kde je soubor, kam se odeslal a co můžeš bezpečně udělat dál?'],['idle','V klidu','Poznáš ihned, kam se nahrávka odešle, a kde začneš nahrávat?'],['recording','Nahrávání','Je jasné, co se právě nahrává, a jak bezpečně skončit?'],['saved','Po nahrávání','Pojmenuj nahrávku a zvol odeslání nebo ponechání na Macu.'],['blocked-company','Chybí firma','Najdi a oprav chybějící firmu. Je jasné, že nahrávka zatím neodešla?'],['uploading','Přehled nahrávek','Rozliš, co je jen na Macu, co se odesílá a co vyžaduje zásah.'],['retry','Chyba a opakování','Zkus znovu odeslat vlastní položku a najdi bezpečné smazání.'],['settings','Nastavení','Najdi firmu, zvuk, automatické odesílání a uchování záznamů.'],['update','Aktualizace','Poznáš přínos nové verze a možnost aktualizaci odložit?']];
const authors={a:{slug:'opus',name:'Claude Opus 5.5'},b:{slug:'astra',name:'Astra'}};
let scenario='idle',theme='light',revealed=true;
const params=new URLSearchParams(location.search);if(scenarios.some(x=>x[0]===params.get('scenario')))scenario=params.get('scenario');if(['light','professional','dark'].includes(params.get('theme')))theme=params.get('theme');
function apply(reload=false){document.querySelector('#comparison').classList.toggle('settings',['uploading','retry','settings','identity','detail'].includes(scenario));document.querySelector('#theme').value=theme;document.querySelector('#task').textContent=scenarios.find(x=>x[0]===scenario)[2];document.querySelectorAll('[data-scenario]').forEach(b=>b.setAttribute('aria-pressed',String(b.dataset.scenario===scenario)));for(const [slot,author] of Object.entries(authors)){const frame=document.querySelector('#frame-'+slot);frame.style.width='';const url='variants/'+author.slug+'/index.html?'+new URLSearchParams({scenario,theme,preview:Date.now()});document.querySelector('#open-'+slot).href='focus.html?'+new URLSearchParams({variant:author.slug,scenario,theme});if(reload||!frame.getAttribute('src'))frame.src=url;else frame.contentWindow.postMessage({type:'ludone-design:set',scenario,theme},location.origin);document.querySelector('#title-'+slot).textContent='Návrh '+slot.toUpperCase()+(revealed?' · '+author.name:'');}history.replaceState(null,'','?'+new URLSearchParams({scenario,theme,preview:Date.now()}));}
for(const [id,label] of scenarios){const b=document.createElement('button');b.textContent=label;b.dataset.scenario=id;b.setAttribute('aria-pressed','false');b.addEventListener('click',()=>{scenario=id;apply(true)});document.querySelector('#scenarios').append(b)}
document.querySelector('#theme').addEventListener('change',e=>{theme=e.target.value;apply(true)});document.querySelector('#reset').addEventListener('click',()=>apply(true));document.querySelector('#reveal').addEventListener('click',e=>{revealed=!revealed;e.target.setAttribute('aria-pressed',String(revealed));e.target.textContent=revealed?'Skrýt autory':'Odhalit autory';for(const [slot,author] of Object.entries(authors))document.querySelector('#title-'+slot).textContent='Návrh '+slot.toUpperCase()+(revealed?' · '+author.name:'')});apply(true);

// Každý návrh může při proklikávání otevřít širší okno nastavení.
window.addEventListener('message',event=>{if(event.origin!==location.origin||event.data?.type!=='ludone-design:state'||!scenarios.some(x=>x[0]===event.data.scenario))return;for(const slot of Object.keys(authors)){const frame=document.querySelector('#frame-'+slot);if(event.source===frame.contentWindow)frame.style.width=['settings','uploading','retry','identity','detail'].includes(event.data.scenario)?'448px':'366px';}});
